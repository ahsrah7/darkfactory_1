# Secret scan patterns for `npm run secrets:scan` (extended regular expressions,
# one per line). Lines starting with # are comments.
# Angle-bracket placeholders such as <user>, <password> and <host> do not match.

# Private key headers
-----BEGIN ([A-Z0-9]+ )?PRIVATE KEY-----

# Cloud access key IDs (AWS)
(AKIA|ASIA)[0-9A-Z]{16}

# Connection strings with embedded credentials (user:password@)
(postgres(ql)?|mysql|mongodb([+]srv)?|redis|amqps?)://[^/[:space:]<:@]+:[^/[:space:]<@][^/[:space:]@]*@

# Generic assignments with a literal value, quoted in code
(password|passwd|secret|token|api[_-]?key)[A-Za-z0-9_-]*["']?[[:space:]]*[:=][[:space:]]*["'][^"'<>[:space:]$]{8,}["']

# Generic assignments with a literal value, unquoted in env-style files
(PASSWORD|SECRET|TOKEN|API_?KEY)[A-Z0-9_]*=[^"'<>[:space:]$]{8,}
